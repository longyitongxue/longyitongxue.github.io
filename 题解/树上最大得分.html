题解

这道题是个树形 DP。由于原图是一棵树，所以天然适合自底向上递推。我们选一个根（比如节点 $1$），对每个节点考虑它的子树。对于节点 $u$，我们只关心它是否和父节点相连，因为度数受父边影响，且儿子之间的选择互不干扰（子树独立）。

我们定义两个 DP 状态，$dp_{u,0}$ 表示在 uuu 的子树内，选一个包含 uuu 的连通块，且 uuu 不连接父节点（即父边被删掉），能得到的最大分；$dp_{u,1}$ 表示在 uuu 的子树内，选一个包含 uuu 的连通块，且 uuu 连接父节点（父边保留），能得到的最大分.

设节点 uuu 有 $m$ 个子节点，将所有子节点 v 的收益 dpv,1 从大到小排序，令 perf(k) 为前 k 个最大收益之和。若 uuu 不连父边，其度数为选中的儿子个数 k，则转移为：

$$ dp_{u,0} = \max_{0 \le k \le m} \big( score_k + \text{pref}(k) \big) $$

若 uuu 连父边，其度数为 $k+1$，则转移为：

$$ dp_{u,1} = \max_{0 \le k \le m} \big( score_{k+1} + \text{pref}(k) \big) $$

最终答案为所有节点 $dp_{u,0}$ 的最大值。

#include<bits/stdc++.h>
#define int int_fast64_t
using namespace std;
int a[int(1e5+5)],n;
vector<int> e[int(1e5+5)];
int dp[int(1e5+5)][2];
void dfs(int u,int fa){
    vector<int> son;
    for(auto v:e[u]){
        if(v==fa) continue;
        dfs(v,u);
        son.push_back(dp[v][1]);
    }
    sort(son.begin(),son.end(),greater<int>());
    dp[u][0]=dp[u][1]=0x8000000000000000;
    int sum=0;
    for(int i=0;i<=son.size();i++){
        dp[u][0]=max(dp[u][0],a[i]+sum);
        if(i+1<n){
            dp[u][1]=max(dp[u][1],a[i+1]+sum);
        }
        if(i<son.size())sum+=son[i];
    }
}
int32_t main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr),cout.tie(nullptr);
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    for(int i=1;i<n;i++){
        int u,v;
        cin>>u>>v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    dfs(1,-1);
    int ans=0x8000000000000000;
    for(int i=1;i<=n;i++){
        ans=max(ans,dp[i][0]);
    }
    cout<<ans;
    return 0;
}
